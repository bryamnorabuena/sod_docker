def set_sod_matrix(job_id, name, description, file_base64, current_user):         
    try:       
        session = SessionLocal()
        time_start = time.perf_counter() 
        now = datetime.now().strftime('%Y-%m-%d %H:%M:%S')                

        matriz = MatrizSap()
        matriz.Nombre = name
        matriz.Descripcion = description
        matriz.IdAppUserCreacion = current_user

        matriz.FechaActualizacion = now

        matriz.Estado = 1

        # Iniciamos la transacción
        session.add(matriz)
        session.flush()

        #storage.update_progress(job_id, 20)

        # Aquí podrías manejar el guardado de archivos si is_new
        base64_zip = file_base64

        zip_bytes = base64.b64decode(base64_zip)
        zip_in_memory = BytesIO(zip_bytes)

        #FALTA CREAR BLOB Y GUARDAR EL ZIP

        response = import_file_access(job_id, zip_in_memory,matriz, session)

        time_end = time.perf_counter()

        if response is True:
            matriz.Tiempo = round(time_end - time_start, 2)
            session.flush()
            session.commit()
            return True
        elif isinstance(response, list):
            session.rollback()            
            #storage.update_job_with_errors(job_id, response)
            return False
        else:
            session.rollback()            
            #storage.update_job_with_errors(job_id, ["Error desconocido al importar la regla."])
            return False
        
    except Exception as e:
        session.rollback()
        # Último intento fallido: notificar al usuario.
        exc_type, exc_obj, exc_tb = sys.exc_info()
        fname = os.path.split(exc_tb.tb_frame.f_code.co_filename)[1]

        error_msg = f"Error en la función set_sod_matrix: {e} en {fname}:{exc_tb.tb_lineno}"
        raise Exception(error_msg)
    except exc.SQLAlchemyError as e:
        raise Exception(f"SQLALCHEMY Error en la función set_sod_matrix: {e}")    
    finally:                
        session.close()

def map_progress(progress_start: int, progress_end: int, p: float) -> int:
        p = max(0.0, min(1.0, p))
        return int(progress_start + p * (progress_end - progress_start))
    
def now_utc_str() -> str:
    return datetime.utcnow().strftime('%Y-%m-%d %H:%M:%S')

def chunked(iterable: Iterable, size: int):
    """Corta un iterable en chunks de tamaño 'size'."""
    chunk = []
    for item in iterable:
        chunk.append(item)
        if len(chunk) >= size:
            yield chunk
            chunk = []
    if chunk:
        yield chunk

def parse_sap_date_or_default(s: str, default: str = "1900-01-01") -> str:  
    """
    Convierte 'dd.mm.yyyy' o 'dd-mm-yyyy' a 'yyyy-mm-dd'. Si está vacío o es inválido, retorna default.
    """
    s = (s or "").strip()
    if not s:
        return default

    # Intenta con dos formatos comunes sin pre-reemplazar el separador.
    for fmt in ("%d.%m.%Y", "%d-%m-%Y"):
        try:
            return datetime.strptime(s, fmt).strftime("%Y-%m-%d")
        except Exception:
            pass

    return default

def bulk_insert_base(session, model_cls, pid: int, nombres: Set[str], matriz, batch_size: int = 5000):
    """
    Inserta en bulk filas base (Rol, Objeto, Autorización, Campo) para 'nombres'.
    No retorna IDs → recarga mapping luego con get_existing_map.
    """
    if not nombres:
        return
    now = now_utc_str()
    batch = []
    for n in nombres:
        batch.append(model_cls(
            IdMatrizSap=pid,
            Nombre=n,
            Descripcion="",
            IdAppUserCreacion=matriz.IdAppUserCreacion,
            IdAppUserActualizacion=matriz.IdAppUserActualizacion,
            FechaCreacion=now,
            FechaActualizacion=now,
            Estado=1
        ))
        if len(batch) >= batch_size:
            session.bulk_save_objects(batch)
            session.flush()
            batch.clear()
    if batch:
        session.bulk_save_objects(batch)
        session.flush()

def get_existing_user_map(
    session,
    UsuarioModel,
    names: Iterable[str],
    *,
    normalize: bool = False,
    active_only: bool = True,
    chunk_size: int = 1000,
) -> Dict[str, int]:
    """
    Devuelve {username: id} para usuarios ya existentes en BD, consultando por bloques.
    - Usa la MISMA sesión que le pasas (no abre otra).
    - 'normalize=True' aplica strip() y lower() a las claves del dict (útil si tu input viene con casing variable).
    - 'active_only=True' filtra por Estado=1 si la columna existe (ajústalo según tu esquema).
    - 'chunk_size' protege contra el límite de parámetros (p.ej. ~2100 en SQL Server).
    """
    if names is None:
        return {}

    # Normalizador (si lo quieres case-insensitive)
    norm = (lambda s: s.strip().lower()) if normalize else (lambda s: s.strip())

    # Conjunto de nombres limpios
    wanted: Set[str] = set()
    for n in names:
        if n is None:
            continue
        s = str(n).strip()
        if s:
            wanted.add(s)

    if not wanted:
        return {}

    result: Dict[str, int] = {}
    col_user = getattr(UsuarioModel, "Usuario")
    col_id   = getattr(UsuarioModel, "Id")
    col_estado = getattr(UsuarioModel, "Estado", None)

    wanted_list = list(wanted)

    for start in range(0, len(wanted_list), chunk_size):
        chunk = wanted_list[start : start + chunk_size]

        q = session.query(col_user, col_id).filter(col_user.in_(chunk))
        if active_only and col_estado is not None:
            q = q.filter(col_estado == 1)

        # Trae (Usuario, Id) y popular dict con la clave normalizada si corresponde
        for uname, uid in q.all():
            result[norm(uname)] = uid if not normalize else uid  # el value no cambia

    return result

def get_existing_map_core(
    session,
    model_cls,
    pid: int,
    names: Optional[Iterable[str]] = None,
    *,
    name_col: str = "Nombre",
    id_col: str   = "Id",
    pid_col: Optional[str] = "IdMatrizSap",
    chunk_size: int = 1000
) -> Dict[str, int]:
    """
    Devuelve {Nombre: Id} para un IdMatrizSap dado usando SQLAlchemy 2.0 (select + execute).
    Si 'names' viene, filtra por esos nombres (con chunking).
    """
    col_name = getattr(model_cls, name_col)
    col_id   = getattr(model_cls, id_col)

    def run_chunk(chunk_names):
        stmt = select(col_name, col_id)
        if pid_col:
            stmt = stmt.where(getattr(model_cls, pid_col) == pid)
        if chunk_names is not None:
            stmt = stmt.where(col_name.in_(chunk_names))
        return session.execute(stmt).all()

    # Sin filtro de nombres (ojo con tamaños grandes)
    if not names:
        rows = run_chunk(None)
        return {n: i for (n, i) in rows}

    wanted: Set[str] = {str(n).strip() for n in names if n is not None and str(n).strip() != ""}
    if not wanted:
        return {}

    result: Dict[str, int] = {}
    wanted_list = list(wanted)

    for start in range(0, len(wanted_list), chunk_size):
        chunk = wanted_list[start : start + chunk_size]
        for n, i in run_chunk(chunk):
            result[n] = i

    return result

def get_user_map(session, names: Optional[Set[str]] = None) -> Dict[str, int]:
    """
    Retorna {Usuario.Usuario -> Id} para todos o solo para 'names'.
    Evita consultas por línea.
    """
    q = session.query(Usuario.Usuario, Usuario.Id)
    if names:
        q = q.filter(Usuario.Usuario.in_(list(names)))
    return {u: i for (u, i) in q.all()}

def get_existing_campo_map(
    session,
    CampoModel,
    id_generica: int,
    names: Optional[Iterable[str]] = None,
    *,
    name_col: str = "Nombre",
    id_col: str   = "Id",
    gen_col: str  = "IdGenerica",
    chunk_size: int = 1000
) -> Dict[str, int]:
    col_name = getattr(CampoModel, name_col)
    col_id   = getattr(CampoModel, id_col)
    col_gen  = getattr(CampoModel, gen_col)

    base_q = session.query(col_name, col_id).filter(col_gen == id_generica)

    if not names:
        return {n: i for (n, i) in base_q.all()}

    wanted: Set[str] = {str(n).strip() for n in names if n is not None and str(n).strip() != ""}
    if not wanted:
        return {}

    result: Dict[str, int] = {}
    wanted_list = list(wanted)
    for start in range(0, len(wanted_list), chunk_size):
        chunk = wanted_list[start : start + chunk_size]
        q = base_q.filter(col_name.in_(chunk))
        for n, i in q.all():
            result[n] = i
    return result
